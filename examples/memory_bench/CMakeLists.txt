cmake_minimum_required(VERSION 3.16)
set(BENCH_PROFILE 2 CACHE STRING "Benchmark profile")
set(ENV{NACBOR_BENCH_PROFILE} "${BENCH_PROFILE}")
if(BENCH_PROFILE GREATER_EQUAL 2 AND NOT BENCH_PROFILE EQUAL 3)
  list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../..")
endif()
if(BENCH_PROFILE GREATER_EQUAL 10 AND BENCH_PROFILE LESS 14)
  list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../../NaWServer")
endif()
if(BENCH_PROFILE LESS 5 OR BENCH_PROFILE EQUAL 14)
  set(LVGL_CBOR_COMPOSITION OFF CACHE BOOL "Nested pages in the benchmark" FORCE)
endif()
if(BENCH_PROFILE LESS 6 OR BENCH_PROFILE EQUAL 14)
  set(LVGL_CBOR_DATA_BINDINGS OFF CACHE BOOL "Data bindings in the benchmark" FORCE)
endif()
if(BENCH_PROFILE LESS 9 OR BENCH_PROFILE EQUAL 14)
  set(LVGL_CBOR_PROJECT_SERVICE OFF CACHE BOOL "ProjectService in the benchmark" FORCE)
endif()
if(BENCH_PROFILE GREATER_EQUAL 8 AND BENCH_PROFILE LESS 14)
  set(LVGL_CBOR_HISTORY ON CACHE BOOL "Optional benchmark history" FORCE)
endif()
if(DEFINED ENV{NACBOR_BENCH_CACHED_COMPONENTS})
  set(ENV{IDF_COMPONENT_MANAGER} 0)
  set(ENV{NA_CBOR_LVGL_COMPONENT} lvgl__lvgl)
  set(ENV{NA_CBOR_CBOR_COMPONENT} espressif__cbor)
  list(APPEND EXTRA_COMPONENT_DIRS "$ENV{NACBOR_BENCH_CACHED_COMPONENTS}/lvgl__lvgl"
    "$ENV{NACBOR_BENCH_CACHED_COMPONENTS}/espressif__cbor")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main __pio_env)
project(nacbor_memory_bench)
if(BENCH_SCREEN_PHASES)
  get_filename_component(core_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  get_filename_component(core_name "${core_root}" NAME)
  idf_component_get_property(bench_core "${core_name}" COMPONENT_LIB)
  idf_component_get_property(bench_main main COMPONENT_LIB)
  target_compile_definitions(${bench_core} PRIVATE LVGL_CBOR_BENCH_SCREEN_PHASES)
  target_compile_definitions(${bench_main} PRIVATE BENCH_SCREEN_PHASES=1)
endif()
if(TARGET __idf_espressif__cbor)
  target_compile_definitions(__idf_espressif__cbor PRIVATE CBOR_PARSER_MAX_RECURSIONS=64)
endif()
if(TARGET __idf_esp_lcd AND CMAKE_HOST_WIN32 AND CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_C_COMPILER_VERSION VERSION_EQUAL "14.2.0")
  target_compile_options(__idf_esp_lcd PRIVATE -Os)
endif()
