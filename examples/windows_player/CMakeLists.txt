cmake_minimum_required(VERSION 3.24)
project(nacborui_windows_player LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This example uses the LVGL Windows backend.")
endif()
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(LVGL_SOURCE_DIR "" CACHE PATH "LVGL 9.6.0 source directory")
set(TINYCBOR_SOURCE_DIR "" CACHE PATH "TinyCBOR source directory (contains src/cbor.h)")
# Reuse downloaded sources when present. No ESP-IDF or board sources are linked.
set(MANAGED "${ROOT}/examples/s3_style_lab/managed_components")
if(NOT LVGL_SOURCE_DIR AND EXISTS "${MANAGED}/lvgl__lvgl/CMakeLists.txt")
  set(LVGL_SOURCE_DIR "${MANAGED}/lvgl__lvgl")
endif()
if(NOT TINYCBOR_SOURCE_DIR AND EXISTS "${MANAGED}/espressif__cbor/tinycbor/src/cbor.h")
  set(TINYCBOR_SOURCE_DIR "${MANAGED}/espressif__cbor/tinycbor")
endif()
if(NOT EXISTS "${LVGL_SOURCE_DIR}/CMakeLists.txt" OR NOT EXISTS "${TINYCBOR_SOURCE_DIR}/src/cbor.h")
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR (LVGL 9.6.0) and TINYCBOR_SOURCE_DIR (TinyCBOR 0.6.1). See README.md; ESP-IDF is not required.")
endif()
option(LVGL_CBOR_GRAPHICS_GALLERY "Include the Windows graphics decoder/vector gallery" ON)
add_compile_definitions(LVGL_CBOR_HAS_GRAPHICS_GALLERY=$<BOOL:${LVGL_CBOR_GRAPHICS_GALLERY}>)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG ${LVGL_CBOR_GRAPHICS_GALLERY} CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL ${LVGL_CBOR_GRAPHICS_GALLERY} CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
set(CBOR "${TINYCBOR_SOURCE_DIR}/src")
add_library(player_cbor STATIC "${CBOR}/cborparser.c" "${CBOR}/cborparser_float.c"
  "${CBOR}/cborvalidation.c" "${CBOR}/cborerrorstrings.c")
target_include_directories(player_cbor PUBLIC "${CBOR}")
target_compile_definitions(player_cbor PRIVATE CBOR_PARSER_MAX_RECURSIONS=64)
add_library(player_runtime STATIC
  "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp" "${ROOT}/src/style_dispatch.cpp"
  "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp"
  "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/capabilities.cpp"
  "${ROOT}/src/project_codec.cpp" "${ROOT}/src/project_service.cpp" "${ROOT}/src/project_store.cpp"
  "${ROOT}/src/project_catalog_store.cpp" "${ROOT}/src/data_value.cpp" "${ROOT}/src/data_registry.cpp" "${ROOT}/src/data_catalog.cpp"
  "${ROOT}/src/binding_expression.cpp" "${ROOT}/src/screen_bindings.cpp" "${ROOT}/src/screen_views.cpp")
target_include_directories(player_runtime PUBLIC "${ROOT}/include")
target_compile_features(player_runtime PUBLIC cxx_std_17)
target_link_libraries(player_runtime PUBLIC lvgl::lvgl player_cbor)
add_executable(nacborui_player src/main.cpp src/application.cpp src/appearance.cpp src/host_fonts.cpp
  src/system_metrics.cpp src/history_plot.cpp src/project_api.cpp src/web_server.cpp src/native_view.cpp ../common/native_view/pointer.cpp src/base_migration.cpp
  ../common/appearance/appearance.cpp ../common/demo_data/demo_data.cpp)
if(LVGL_CBOR_GRAPHICS_GALLERY)
  target_sources(nacborui_player PRIVATE src/graphics_gallery.cpp ../common/polygon_button/polygon_button.cpp)
  target_include_directories(nacborui_player PRIVATE "${ROOT}/examples/common/polygon_button")
  file(READ "${ROOT}/examples/common/polygon_button/catalog.json" POLYGON_BUTTON_CATALOG_JSON)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/examples/common/polygon_button/catalog.json")
  configure_file("${ROOT}/examples/common/polygon_button/catalog.hpp.in" "${CMAKE_CURRENT_BINARY_DIR}/polygon_button_catalog.hpp" @ONLY)
endif()
target_compile_definitions(nacborui_player PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
file(READ "${CMAKE_CURRENT_LIST_DIR}/history-plot.json" HISTORY_PLOT_CATALOG)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/history-plot.json")
configure_file(src/history_plot_catalog.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/history_plot_catalog.hpp" @ONLY)
target_include_directories(nacborui_player PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nacborui_player PRIVATE player_runtime user32 gdi32 imm32 shcore comdlg32 pdh dxgi)
if(MSVC)
  target_compile_options(nacborui_player PRIVATE /utf-8 /W4)
endif()

option(LVGL_CBOR_HISTORY "Include optional registry history in the Windows example" ON)
target_compile_definitions(player_runtime PUBLIC LVGL_CBOR_HAS_HISTORY=$<BOOL:${LVGL_CBOR_HISTORY}>)
if(LVGL_CBOR_HISTORY)
  target_sources(player_runtime PRIVATE "${ROOT}/src/data_history.cpp")
endif()

set(NAWSERVER_SOURCE_DIR "" CACHE PATH "NaWServer source directory (native Windows backend)")
if(NOT NAWSERVER_SOURCE_DIR)
  if(EXISTS "${ROOT}/../NaWServer/cmake/NaWServer.cmake")
    set(NAWSERVER_SOURCE_DIR "${ROOT}/../NaWServer")
  elseif(EXISTS "${ROOT}/local/deps/NaWServer/cmake/NaWServer.cmake")
    set(NAWSERVER_SOURCE_DIR "${ROOT}/local/deps/NaWServer")
  endif()
endif()
if(NOT EXISTS "${NAWSERVER_SOURCE_DIR}/cmake/NaWServer.cmake")
  message(FATAL_ERROR "Set NAWSERVER_SOURCE_DIR to the NaWServer checkout")
endif()
include("${NAWSERVER_SOURCE_DIR}/cmake/NaWServer.cmake")
nawserver_add_library(player_webserver PLATFORM windows MANAGED NO_SSE NO_AUTH NO_CLIENT_QUOTAS NO_API_KEYS)
target_link_libraries(nacborui_player PRIVATE player_webserver)

# Optional proof of concept. Ordinary builds contain no MCP/JSON parser workspace.
option(LVGL_CBOR_NATIVE_MCP "Compile the optional native NaWServer MCP proof of concept" OFF)
target_compile_definitions(nacborui_player PRIVATE LVGL_CBOR_NATIVE_MCP=$<BOOL:${LVGL_CBOR_NATIVE_MCP}>)
if(LVGL_CBOR_NATIVE_MCP)
  include("${NAWSERVER_SOURCE_DIR}/cmake/NaWServerJson.cmake")
  nawserver_attach_json(player_webserver bounded)
  target_sources(player_webserver PRIVATE "${NAWSERVER_SOURCE_DIR}/src/mcp/mcp.cpp")
  target_compile_definitions(player_webserver PUBLIC NAWSERVER_MCP=1
    NAWSERVER_JSON_INPUT_BYTES=65535 NAWSERVER_JSON_STRING_BYTES=65535
    NAWSERVER_JSON_CONTAINER_ITEMS=4096 NAWSERVER_JSON_DEPTH=32
    NAWSERVER_JSON_VALUE_CAPACITY=8192 NAWSERVER_JSON_WORKSPACE_BYTES=65536
    NAWSERVER_MCP_OUTPUT_BYTES=65535)
  target_sources(nacborui_player PRIVATE "${ROOT}/integrations/nawserver/native_mcp.cpp")
  target_include_directories(nacborui_player PRIVATE "${ROOT}/integrations/nawserver")
endif()

target_sources(nacborui_player PRIVATE ../common/layout_debug/layout_diagnostics.cpp)
target_include_directories(nacborui_player PRIVATE "${ROOT}/examples/common/layout_debug")

# Verify owned large-response lifetime independently of device hardware.
add_executable(nacborui_http_payload_tests "${ROOT}/tests/http_payload_test.cpp")
target_include_directories(nacborui_http_payload_tests PRIVATE "${ROOT}/integrations/nawserver")
target_link_libraries(nacborui_http_payload_tests PRIVATE player_webserver)
target_compile_definitions(nacborui_http_payload_tests PRIVATE CONFIG_NAWSERVER_TCP_MANAGED=1)
if(MSVC)
  target_compile_options(nacborui_http_payload_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(nacborui_http_payload_tests PRIVATE -UNDEBUG)
endif()

add_executable(nacborui_input_queue_tests "${ROOT}/tests/native_input_queue_test.cpp")
target_include_directories(nacborui_input_queue_tests PRIVATE "${ROOT}/examples/common/native_view")
if(MSVC)
 target_compile_options(nacborui_input_queue_tests PRIVATE /UNDEBUG)
else()
 target_compile_options(nacborui_input_queue_tests PRIVATE -UNDEBUG)
endif()
