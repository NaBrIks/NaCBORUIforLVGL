cmake_minimum_required(VERSION 3.24)
project(nacborui_mcp_gateway LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This test/development gateway uses the Windows NaWServer backend")
endif()
get_filename_component(ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(NAWSERVER_SOURCE_DIR "${ROOT}/../NaWServer" CACHE PATH "NaWServer source")
include("${NAWSERVER_SOURCE_DIR}/cmake/NaWServer.cmake")
nawserver_add_library(gateway_web PLATFORM windows MANAGED NO_SSE NO_AUTH NO_CLIENT_QUOTAS NO_API_KEYS)
add_executable(nacborui_mcp_gateway main.cpp)
target_link_libraries(nacborui_mcp_gateway PRIVATE gateway_web)
target_compile_features(nacborui_mcp_gateway PRIVATE cxx_std_17)
target_compile_definitions(nacborui_mcp_gateway PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
 target_compile_options(nacborui_mcp_gateway PRIVATE /utf-8 /W4)
endif()
