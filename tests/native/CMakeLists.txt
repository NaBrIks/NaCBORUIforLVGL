cmake_minimum_required(VERSION 3.24)
project(style_runtime_tests LANGUAGES C CXX)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(DEPS "${ROOT}/examples/s3_style_lab/managed_components")
set(LVGL_SOURCE_DIR "${DEPS}/lvgl__lvgl" CACHE PATH "LVGL 9.6.0 source directory")
set(TINYCBOR_SOURCE_DIR "${DEPS}/espressif__cbor/tinycbor" CACHE PATH "TinyCBOR 0.6.1 source directory")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/CMakeLists.txt" OR NOT EXISTS "${TINYCBOR_SOURCE_DIR}/src/cbor.h")
  message(FATAL_ERROR "Set LVGL_SOURCE_DIR and TINYCBOR_SOURCE_DIR; see Readme.md native-test setup.")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
set(CBOR "${TINYCBOR_SOURCE_DIR}/src")
add_library(cbor STATIC "${CBOR}/cborparser.c" "${CBOR}/cborparser_float.c" "${CBOR}/cborvalidation.c" "${CBOR}/cborerrorstrings.c")
target_include_directories(cbor PUBLIC "${CBOR}")
target_compile_definitions(cbor PRIVATE CBOR_PARSER_MAX_RECURSIONS=64)
add_executable(style_runtime_tests main.cpp "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp" "${ROOT}/src/style_dispatch.cpp")
target_compile_features(style_runtime_tests PRIVATE cxx_std_17)
target_include_directories(style_runtime_tests PRIVATE "${ROOT}/include")
target_link_libraries(style_runtime_tests PRIVATE lvgl::lvgl cbor)
add_executable(screen_runtime_tests screens.cpp "${ROOT}/src/capabilities.cpp" "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp" "${ROOT}/examples/s3_style_lab/main/component_preview.cpp")
target_compile_features(screen_runtime_tests PRIVATE cxx_std_17)
option(LVGL_CBOR_NAIVE_EMBEDDED "Test the experimental reader for trusted embedded screens" OFF)
if(LVGL_CBOR_NAIVE_EMBEDDED)
  target_compile_definitions(screen_runtime_tests PRIVATE LVGL_CBOR_NAIVE_EMBEDDED)
endif()
target_include_directories(screen_runtime_tests PRIVATE "${ROOT}/include")
target_link_libraries(screen_runtime_tests PRIVATE lvgl::lvgl cbor)

add_executable(style_service_tests services.cpp "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/capabilities.cpp")
target_compile_features(style_service_tests PRIVATE cxx_std_17)
target_include_directories(style_service_tests PRIVATE "${ROOT}/include")
target_link_libraries(style_service_tests PRIVATE lvgl::lvgl cbor)

add_executable(native_widget_tests native_widgets.cpp "${ROOT}/examples/s3_style_lab/main/native_widgets/registry.cpp")
target_link_libraries(native_widget_tests PRIVATE lvgl::lvgl cbor)
target_compile_features(native_widget_tests PRIVATE cxx_std_17)
target_include_directories(native_widget_tests PRIVATE "${ROOT}/include")
target_sources(native_widget_tests PRIVATE "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp" "${ROOT}/src/capabilities.cpp")
target_sources(native_widget_tests PRIVATE "${ROOT}/src/project_codec.cpp" "${ROOT}/src/project_service.cpp" "${ROOT}/src/project_store.cpp" "${ROOT}/src/project_catalog_store.cpp")
add_executable(demo_speed_tests demo_speed.cpp)
target_compile_features(demo_speed_tests PRIVATE cxx_std_17)
add_executable(data_events_tests data_events.cpp "${ROOT}/examples/s3_style_lab/main/data_events.cpp" "${ROOT}/src/data_value.cpp" "${ROOT}/src/data_registry.cpp" "${ROOT}/src/data_catalog.cpp")
target_compile_features(data_events_tests PRIVATE cxx_std_17)
target_include_directories(data_events_tests PRIVATE "${ROOT}/include")
target_link_libraries(data_events_tests PRIVATE cbor)

add_executable(native_widget_disabled_tests native_widgets_disabled.cpp "${ROOT}/examples/s3_style_lab/main/native_widgets/registry.cpp" "${ROOT}/src/widget_registry.cpp")
target_compile_definitions(native_widget_disabled_tests PRIVATE LVGL_CBOR_DEMO_NATIVE_WIDGETS=0)
target_compile_features(native_widget_disabled_tests PRIVATE cxx_std_17)
target_include_directories(native_widget_disabled_tests PRIVATE "${ROOT}/include")
target_link_libraries(native_widget_disabled_tests PRIVATE lvgl::lvgl cbor)

add_executable(project_service_tests projects.cpp "${ROOT}/src/project_codec.cpp" "${ROOT}/src/project_service.cpp" "${ROOT}/src/project_store.cpp" "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp")
target_compile_features(project_service_tests PRIVATE cxx_std_17)
target_sources(project_service_tests PRIVATE "${ROOT}/src/project_catalog_store.cpp")
target_include_directories(project_service_tests PRIVATE "${ROOT}/include")
target_link_libraries(project_service_tests PRIVATE lvgl::lvgl cbor)

set(BINDING_SOURCES "${ROOT}/src/data_value.cpp" "${ROOT}/src/data_registry.cpp" "${ROOT}/src/data_catalog.cpp" "${ROOT}/src/binding_expression.cpp" "${ROOT}/src/screen_bindings.cpp" "${ROOT}/src/screen_views.cpp")
foreach(target screen_runtime_tests style_service_tests native_widget_tests project_service_tests)
  target_sources(${target} PRIVATE ${BINDING_SOURCES})
endforeach()
add_executable(system_status_tests system_status.cpp)
target_compile_features(system_status_tests PRIVATE cxx_std_17)
add_executable(binding_tests bindings.cpp ${BINDING_SOURCES} "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp")
target_compile_features(binding_tests PRIVATE cxx_std_17)
target_include_directories(binding_tests PRIVATE "${ROOT}/include")
target_link_libraries(binding_tests PRIVATE lvgl::lvgl cbor)
add_executable(composition_tests compositions.cpp ${BINDING_SOURCES} "${ROOT}/src/project_codec.cpp" "${ROOT}/src/project_service.cpp" "${ROOT}/src/project_store.cpp" "${ROOT}/src/project_catalog_store.cpp" "${ROOT}/src/screen_runtime.cpp" "${ROOT}/src/screen_decode.cpp" "${ROOT}/src/widget_registry.cpp" "${ROOT}/src/widget_dispatch.cpp" "${ROOT}/src/screen_styles.cpp" "${ROOT}/src/style_dispatch.cpp" "${ROOT}/src/style_service.cpp" "${ROOT}/src/style_runtime.cpp")
target_compile_features(composition_tests PRIVATE cxx_std_17)
target_include_directories(composition_tests PRIVATE "${ROOT}/include")
target_link_libraries(composition_tests PRIVATE lvgl::lvgl cbor)

add_executable(data_catalog_tests data_catalog.cpp "${ROOT}/src/data_value.cpp" "${ROOT}/src/data_registry.cpp" "${ROOT}/src/data_catalog.cpp")
target_compile_features(data_catalog_tests PRIVATE cxx_std_17)
target_include_directories(data_catalog_tests PRIVATE "${ROOT}/include")

target_compile_features(data_history_tests PRIVATE cxx_std_17)
target_compile_definitions(data_history_tests PRIVATE LVGL_CBOR_HAS_HISTORY=1)
